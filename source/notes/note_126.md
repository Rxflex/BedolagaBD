```json
{
  "tag": "auto-lte",
  "selector": ["lte", "proxy"],
  "strategy": {
    "type": "leastLoad",
    "settings": {
      "costs": [
        {"match": "lte-lc", "value": 1000},
        {"match": "proxy", "value": 1}
      ],
      "maxRTT": "1s",
      "tolerance": 0.35
    }
  },
  "fallbackTag": "direct"
}
```
