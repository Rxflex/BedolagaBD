  ```bash
  # Бэкап:
  cp app/external/remnawave_api.py app/external/remnawave_api.py.bak

  # Патч (одна строка):
  sed -i "s/uuid=user_data\['uuid'\],/uuid=str(user_data['id']),/" app/external/remnawave_api.py

  # Убедиться что заменилось (должна быть ровно одна строка):
  grep -n "uuid=str(user_data\['id'\])" app/external/remnawave_api.py

  # Пересобрать:
  docker compose up -d --build bot
  sleep 15

  # После всех проверок можно закрепить
  cd /opt/bedolaga
  git checkout -b local/remnawave-3.0-compat
  git add app/external/remnawave_api.py
  git commit -m "patch: use numeric id as uuid for Remnawave 3.0 compat"
  ```
